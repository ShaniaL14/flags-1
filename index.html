World Flag Explorer

flags = transpose(flag_data)

// Dropdown input to select a country
viewof selected = Inputs.select(flags, {
  label: "Choose a Country:",
  format: d => d.name
})

// Function to switch selected country when a card is clicked
function selectFlag(code) {
  const match = flags.find(d => d.code === code)
  if (match) {
    viewof selected.value = match
    viewof selected.dispatchEvent(new CustomEvent("input"))
  }
}

similarFlags = selected.similar.code.map((c, i) => ({
  code: c,
  name: selected.similar.name[i]
}))

differentFlags = selected.different.code.map((c, i) => ({
  code: c,
  name: selected.different.name[i]
}))
